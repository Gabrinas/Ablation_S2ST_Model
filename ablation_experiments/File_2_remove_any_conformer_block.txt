class ConformerEncoder(nn.Module):
    def __init__(self, num_layers, d_model, d_ff, num_heads,
                 kernel_size=31, dropout=0.1, input_dim=None,
                 remove_blocks=None):
        super().__init__()
        # Input projection
        if input_dim is not None and input_dim != d_model:
            self.input_proj = nn.Linear(input_dim, d_model)
        else:
            self.input_proj = nn.Identity()

        # Default: no blocks removed
        if remove_blocks is None:
            remove_blocks = []

        # Build stack with selective removal
        self.layers = nn.ModuleList([
            ConformerBlock(d_model, d_ff, num_heads, kernel_size, dropout)
            if i not in remove_blocks else nn.Identity()
            for i in range(num_layers)
        ])

        self.final_norm = nn.LayerNorm(d_model)

    def forward(self, x, mask=None):
        x = self.input_proj(x)
        for layer in self.layers:
            # Identity just returns x unchanged
            x = layer(x, mask) if not isinstance(layer, nn.Identity) else layer(x)
        return self.final_norm(x)
